# yadm bootstrap##os.Darwin, the Mac's entry point (Story 2.61), run end to end
# against a scratch HOME and a scratch repo with real yadm and git.
# BOOTSTRAP_TEST_MODE=1 passes the Darwin guard and lets a pipe answer the
# decrypt prompt; `yadm config local.os Darwin` makes yadm link the Darwin
# alternates on Linux CI. A yadm wrapper fakes only `yadm decrypt`, since the
# real one needs the archive passphrase. `mise bootstrap` (step 6) installs
# system-wide packages, so every run passes --no-mise; the manual evidence for
# it is `mise bootstrap --dry-run` in the runbook.
#
# Run from the repo root: clitest tests/mac-bootstrap.clitest.txt

$ B=$PWD/.config/yadm/bootstrap##os.Darwin; RULES=$PWD/.config/yadm/sparse/work-mac; T=$(mktemp -d); export GIT_CONFIG_GLOBAL=/dev/null GIT_CONFIG_NOSYSTEM=1 GIT_AUTHOR_NAME=t GIT_AUTHOR_EMAIL=t@example.invalid GIT_COMMITTER_NAME=t GIT_COMMITTER_EMAIL=t@example.invalid; echo ok
ok
$ S=$T/src; mkdir -p "$S/.config/i3" "$S/.config/mise" "$S/.config/yadm/sparse" "$S/.local/share/yadm" && echo 'repo zshrc' > "$S/.zshrc" && echo linux > "$S/.config/i3/config" && echo 'darwin mise' > "$S/.config/mise/config.toml##os.Darwin" && echo '[credential]' > "$S/.gitconfig-os##os.Darwin" && cp "$RULES" "$S/.config/yadm/sparse/work-mac" && echo ciphertext > "$S/.local/share/yadm/archive" && git init -q -b main "$S" && git -C "$S" add -A && git -C "$S" commit -qm seed && echo seeded
seeded
$ export HOME=$T/home XDG_CONFIG_HOME=$T/home/.config XDG_DATA_HOME=$T/home/.local/share XDG_STATE_HOME=$T/home/.local/state; mkdir -p "$HOME/.config/mise" "$HOME/.ssh" && echo mine > "$HOME/.zshrc" && echo 'live mise' > "$HOME/.config/mise/config.toml" && echo 'my ssh' > "$HOME/.ssh/config" && echo home
home
$ REAL=$(command -v yadm); mkdir "$T/bin" && printf '%s\n' '#!/bin/sh' 'if [ "$1" = decrypt ]; then' '  if [ "$2" = -l ]; then printf "%s\n" .ssh/config .zshenv; exit 0; fi' '  echo run >> "$T/decrypted"; echo "from archive" > "$HOME/.ssh/config"; exit 0' 'fi' "exec $REAL \"\$@\"" > "$T/bin/yadm" && chmod +x "$T/bin/yadm"; export PATH="$T/bin:$PATH" T; echo wrapper
wrapper
$ cd "$HOME" && yadm config yadm.auto-alt false && yadm clone -q --no-checkout --no-bootstrap "$S" >/dev/null 2>&1 && yadm config local.os Darwin && echo cloned
cloned
$ snap() { (cd "$HOME" && find . -path ./.local/share/yadm -prune -o -print | LC_ALL=C sort | while IFS= read -r p; do if [ -L "$p" ]; then echo "$p -> $(readlink "$p")"; elif [ -f "$p" ]; then echo "$p $(cksum < "$p")"; else echo "$p/"; fi; done; git --git-dir="$XDG_DATA_HOME/yadm/repo.git" config --get-regexp 'local\.class|core\.sparsecheckout'); }; snap > "$T/before"; echo snapped
snapped
$
# Without the test switch, Linux stops at the guard
$ printf '%s\n' '#!/bin/sh' 'echo Linux' > "$T/bin/uname"; chmod +x "$T/bin/uname"; bash "$B" --check 2>&1; echo "rc=$?"; rm "$T/bin/uname"
bootstrap: FATAL: this bootstrap is for macOS; Linux uses .local/bin/setup/bootstrap
rc=1
