# setup/shell-caches — the per-user files .zshrc sources but the repo can't
# track. Host-independent: a scratch HOME and stub tools on PATH; nothing is
# cloned or installed.
#
# Run from the repo root: clitest tests/shell-caches.clitest.txt

$ T=$(mktemp -d); mkdir -p "$T/bin" "$T/tools" "$T/home"; for t in bash sh sed cat mkdir mv rm tr wc touch chmod mktemp; do ln -s "$(command -v $t)" "$T/tools/$t"; done; printf '%s\n' '#!/bin/sh' "PATH=$T/bin:$T/tools exec $PWD/.local/bin/setup/shell-caches \"\$@\"" > "$T/sc"; chmod +x "$T/sc"; SC=$T/sc; export HOME=$T/home; echo ok
ok
$
# no tool installed: every part is skipped and nothing is missing
$ "$SC" --check; echo "rc=$?"
  antibody: not installed; skipped
rc=0
$
# antibody: the stub "clones" by creating the cache file the bundle sources
$ printf '%s\n' '#!/bin/sh' 'cat > /dev/null' 'echo clone >> "$HOME/antibody.log"' '[ -f "$HOME/offline" ] && exit 1' 'mkdir -p "$HOME/.cache/antibody/zsh-syntax-highlighting"' ': > "$HOME/.cache/antibody/zsh-syntax-highlighting/plugin.zsh"' 'echo "source $HOME/.cache/antibody/zsh-syntax-highlighting/plugin.zsh"' > "$T/bin/antibody"; chmod +x "$T/bin/antibody"; echo zsh-users/zsh-syntax-highlighting > "$HOME/.zsh_plugins.txt"; echo "source $HOME/.cache/antibody/zsh-syntax-highlighting/plugin.zsh" > "$HOME/.zsh_plugins.sh"; echo stub
stub
$ "$SC" --check | sed "s|$T|T|g"; echo "rc=${PIPESTATUS[0]}"; test -e "$HOME/antibody.log" || echo "nothing cloned"
  antibody: would run: antibody bundle < ~/.zsh_plugins.txt
rc=1
nothing cloned
$
# offline: apply warns and still exits 0, so bootstrap goes on
$ touch "$HOME/offline"; "$SC" 2>&1; echo "rc=$?"; rm "$HOME/offline"
  antibody: bundle failed (offline?); re-run shell-caches later
rc=0
$
# apply clones, keeps the tracked bundle as it is, and a second run is a no-op
$ "$SC"; "$SC"; "$SC" --check; echo "rc=$?"; wc -l < "$HOME/antibody.log" | tr -d ' '; sed "s|$T|T|g" "$HOME/.zsh_plugins.sh"
  antibody: cloned the plugins
  antibody: plugins in place
  antibody: plugins in place
rc=0
2
source T/home/.cache/antibody/zsh-syntax-highlighting/plugin.zsh
$
# a bundle naming paths antibody did not create (another user's HOME) is reported
$ echo "source /nonexistent/plugin.zsh" > "$HOME/.zsh_plugins.sh"; "$SC" 2>&1; echo "rc=$?"
  antibody: bundled, but ~/.zsh_plugins.sh names paths still missing; regenerate it: antibody bundle < ~/.zsh_plugins.txt > ~/.zsh_plugins.sh
rc=0
$
# unknown argument fails loudly
$ "$SC" frobnicate 2>&1; echo "rc=$?"
shell-caches: unknown argument: frobnicate   (try: shell-caches -h)
rc=2
$ rm -rf "$T"; echo done
done
$
