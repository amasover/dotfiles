# Hibernate pre-freeze eviction. No root, real sysfs, or sleep calls.
# Run from repo root: clitest tests/hibernate-prereclaim.clitest.txt

$ tool=$PWD/.config/dotfiles/hibernate/hibernate-prereclaim; R=$(mktemp -d); export HIBERNATE_PRERECLAIM_ROOT=$R; mkdir -p "$R/sys/module/zswap/parameters" "$R/sys/fs/cgroup" "$R/sys/power" "$R/proc" "$R/run/hibernate-prereclaim"; p=$R/sys/module/zswap/parameters; state=$R/run/hibernate-prereclaim/previous; knob=$R/sys/fs/cgroup/memory.reclaim; loaded() { printf 'MemTotal:        8388608 kB\nMemFree:         1048576 kB\nZswap:            524288 kB\n' > "$R/proc/meminfo"; }; printf '3221225472\n' > "$R/sys/power/image_size"; loaded
$
$ reset() { printf 'Y\n' > "$p/enabled"; printf 'N\n' > "$p/shrinker_enabled"; : > "$knob"; loaded; rm -f "$state"; }
$

Route the eviction past the pool and restore the original setting afterwards.
The shrinker setting is not ours to touch: writeback never triggered during a
measured 1.5 GB reclaim, so it stays exactly as the machine had it:

$ reset; "$tool" prepare; cat "$p/enabled" "$p/shrinker_enabled"
hibernate-prereclaim: evicted 4508876 kB of 4508876 kB requested
hibernate-prereclaim: zswap pool 524288 kB before, 524288 kB after
N
N
$ "$tool" restore; cat "$p/enabled" "$p/shrinker_enabled"; test ! -e "$state"; echo "rc=$?"
hibernate-prereclaim: restored zswap enabled=Y
Y
N
rc=0
$

A shrinker the machine had switched on stays on across the whole operation:

$ reset; printf 'Y\n' > "$p/shrinker_enabled"; "$tool" prepare > /dev/null; cat "$p/shrinker_enabled"; "$tool" restore > /dev/null; cat "$p/enabled" "$p/shrinker_enabled"
Y
Y
Y
$

Ask for nothing when the footprint already fits under the image target:

$ reset; printf 'MemTotal:        8388608 kB\nMemFree:         8000000 kB\nZswap:                 0 kB\n' > "$R/proc/meminfo"; "$tool" prepare
hibernate-prereclaim: footprint already under image target
hibernate-prereclaim: zswap pool 0 kB before, 0 kB after
$ "$tool" restore > /dev/null; cat "$p/enabled"
Y
$

A kernel that stops reclaiming ends the eviction without failing the hibernate:

$ reset; rm "$knob"; mkdir "$knob"; "$tool" prepare; echo "rc=$?"
hibernate-prereclaim: reclaim stalled after 0 of 4508876 kB
hibernate-prereclaim: zswap pool 524288 kB before, 524288 kB after
rc=0
$ rmdir "$knob"; "$tool" restore > /dev/null; cat "$p/enabled"
Y
$

Missing kernel interfaces skip the eviction rather than block hibernation:

$ reset; rm "$knob"; "$tool" prepare; echo "rc=$?"
hibernate-prereclaim: kernel interfaces unavailable; skipped eviction
hibernate-prereclaim: zswap pool 524288 kB before, 524288 kB after
rc=0
$ "$tool" restore > /dev/null; cat "$p/enabled"
Y
$

The same restore operation recovers after a failed sleep operation:

$ reset; (trap '"$tool" restore > /dev/null' EXIT; "$tool" prepare > /dev/null; exit 42); echo "rc=$?"; cat "$p/enabled"
rc=42
Y
$

A second prepare must not overwrite the original saved value:

$ reset; "$tool" prepare > /dev/null; if "$tool" prepare > /dev/null 2>&1; then echo unexpected-success; else echo rejected; fi; "$tool" restore > /dev/null; cat "$p/enabled"
rejected
Y
$

Reject unknown kernel state without changing it, including failure cleanup:

$ reset; printf 'unexpected\n' > "$p/enabled"; "$tool" prepare > /dev/null 2>&1; echo "rc=$?"; cat "$p/enabled"; "$tool" restore; test ! -e "$state"; echo "rc=$?"
rc=1
unexpected
rc=0
$

Reject a saved parameter this tool never writes, rather than poking sysfs:

$ reset; printf 'max_pool_percent=1\n' > "$state"; "$tool" restore 2>&1; echo "rc=$?"; test -f "$state"; echo "kept=$?"
hibernate-prereclaim: unknown saved parameter
rc=1
kept=0
$

Retain the saved value if restoration fails, so restoration can be retried:

$ reset; "$tool" prepare > /dev/null; rm "$p/enabled"; mkdir "$p/enabled"; "$tool" restore > /dev/null 2>&1; test -f "$state"; echo "saved=$?"; rmdir "$p/enabled"; printf 'N\n' > "$p/enabled"; "$tool" restore > /dev/null; cat "$p/enabled"; test ! -e "$state"; echo "cleared=$?"
saved=0
Y
cleared=0
$

Reject a deadline that is not a whole number of seconds:

$ reset; HIBERNATE_PRERECLAIM_DEADLINE=soon "$tool" prepare 2>&1; echo "rc=$?"
hibernate-prereclaim: deadline must be a whole number of seconds
rc=2
$

Reject a relative root rather than guessing a prefix:

$ HIBERNATE_PRERECLAIM_ROOT=relative "$tool" prepare 2>&1; echo "rc=$?"
hibernate-prereclaim: root must be absolute
rc=2
$ rm -rf "$R"; unset HIBERNATE_PRERECLAIM_ROOT
$

The installer is the one file list the runbook, README and rehearsal share. It
also evicts the retired shrinker workaround from machines that still carry it:

$ installer=$PWD/.config/dotfiles/hibernate/install; I=$(mktemp -d); export HIBERNATE_PRERECLAIM_ROOT=$I; d=$I/etc/systemd/system/systemd-hibernate.service.d; mkdir -p "$d" "$I/usr/local/libexec"; : > "$d/50-zswap-shrinker.conf"; : > "$I/usr/local/libexec/hibernate-zswap-shrinker"; : > "$d/90-unrelated.conf"
