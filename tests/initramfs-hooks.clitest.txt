# Initramfs HOOKS policy link, bootstrap step 6e (Story 2.59, #258).
# Run from repo root: clitest tests/initramfs-hooks.clitest.txt
#
# Host-independent by construction: HOME, the /etc and /proc/cmdline prefix, and
# every privileged binary are redirected at a scratch tree, so this needs no root
# and never touches the running machine's initramfs.

$ ITMP=$(mktemp -d); IHOME=$ITMP/home; IROOT=$ITMP/root; IBIN=$ITMP/bin; SRC=$IHOME/.config/dotfiles/mkinitcpio/hooks.conf; DST=$IROOT/etc/mkinitcpio.conf.d/10-dotfiles-hooks.conf; TRACE=$ITMP/trace; export BOOTSTRAP_TEST_MODE=1 TRACE; mkdir -p "${SRC%/*}" "$IROOT/etc" "$IROOT/proc" "$IBIN"; cp .config/dotfiles/mkinitcpio/hooks.conf "$SRC"; echo ready
ready
$ printf '%s\n' '#!/bin/sh' 'if [ "$1" = mkinitcpio ]; then echo "$*" >> "$TRACE"; exit 0; fi' 'exec "$@"' > "$IBIN/sudo"; chmod +x "$IBIN/sudo"; echo stubs
stubs
$
