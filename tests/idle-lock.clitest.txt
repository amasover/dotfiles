hibernate plus an uninhibited hard chain for laptop hibernation at 90 minutes:

$ XIDLEHOOK_LOG_DIR="$CLITMP/xidlehook" XDG_RUNTIME_DIR="$CLITMP/runtime" HOME=/fixture PATH="$CLITMP/bin:$PATH" $idle & launcher=$!; i=0; while { [ ! -s "$CLITMP/xidlehook/soft" ] || [ ! -s "$CLITMP/xidlehook/hard" ]; } && [ "$i" -lt 50 ]; do sleep 0.02; i=$((i + 1)); done; kill "$launcher" 2>/dev/null || :; wait "$launcher" 2>/dev/null || :; for name in soft hard; do [ -s "$CLITMP/xidlehook/$name" ] && echo "$name-started"; done
