# Hibernate-only zswap shrinker transitions. No root, real sysfs, or sleep calls.
# Run from repo root: clitest tests/hibernate-zswap.clitest.txt

$ tool=$PWD/.config/dotfiles/hibernate/hibernate-zswap-shrinker; R=$(mktemp -d); export HIBERNATE_ZSWAP_ROOT=$R; mkdir -p "$R/sys/module/zswap/parameters" "$R/run/hibernate-zswap-shrinker"; knob=$R/sys/module/zswap/parameters/shrinker_enabled; state=$R/run/hibernate-zswap-shrinker/previous; printf 'Y\n' > "$knob"; printf 'Y\n' > "$R/sys/module/zswap/parameters/enabled"
$

Pause proactive shrinking, but not zswap; restore the original setting:

$ "$tool" prepare >/dev/null; cat "$knob" "$R/sys/module/zswap/parameters/enabled"
N
Y
$ "$tool" restore >/dev/null; cat "$knob"; test ! -e "$state"; echo "rc=$?"
Y
rc=0
$

Do not enable a shrinker that was already disabled:

$ printf 'N\n' > "$knob"; "$tool" prepare >/dev/null; "$tool" restore >/dev/null; cat "$knob"
N
$

The same restore operation recovers after a failed sleep operation:

$ printf 'Y\n' > "$knob"; (trap '"$tool" restore >/dev/null' EXIT; "$tool" prepare >/dev/null; exit 42); echo "rc=$?"; cat "$knob"
rc=42
Y
$

A second prepare must not overwrite the original saved value:

$ "$tool" prepare >/dev/null; if "$tool" prepare >/dev/null 2>&1; then echo unexpected-success; else echo rejected; fi; "$tool" restore >/dev/null; cat "$knob"
rejected
Y
$

Reject unknown kernel state without changing it, including failure cleanup:

$ printf 'unexpected\n' > "$knob"; "$tool" prepare >/dev/null 2>&1; echo "rc=$?"; "$tool" restore >/dev/null; cat "$knob"; test ! -e "$state"; echo "rc=$?"
rc=1
unexpected
rc=0
$

Retain the saved value if restoration fails, so restoration can be retried:

$ printf 'Y\n' > "$knob"; "$tool" prepare >/dev/null; rm "$knob"; mkdir "$knob"; "$tool" restore >/dev/null 2>&1; test -f "$state"; echo "saved=$?"; rmdir "$knob"; printf 'N\n' > "$knob"; "$tool" restore >/dev/null; cat "$knob"; test ! -e "$state"; echo "cleared=$?"
saved=0
Y
cleared=0
$ rm -rf "$R"; unset HIBERNATE_ZSWAP_ROOT
$
