# setup/vim-plugins — packaged manager plus fallback apply/update/check behavior
# (Story 2.49). Host-independent: fake Vim mutates only $CLITMP; no network or
# live ~/.vim state is used.
#
# Run from the repo root: clitest tests/vim-plugins.clitest.txt

$ VP=.local/bin/setup/vim-plugins; CLITMP=$(mktemp -d); mkdir -p "$CLITMP/bin" "$CLITMP/runtime/autoload"; printf "%s\n" "Plug 'owner/foo'" "\"Plug 'ignored/nope'" "Plug 'owner/bar', {'rtp': 'vim/'}" > "$CLITMP/vimrc"; touch "$CLITMP/runtime/autoload/plug.vim"; RUN() { PATH="$CLITMP/bin:$PATH" FAKE_LOG="$CLITMP/call" VIM_PLUGIN_HOME="$CLITMP/vim" VIM_PLUG_RUNTIME="$CLITMP/runtime/autoload/plug.vim" "$VP" "$@"; }; echo ready
ready
$
# list derives vim-plug's default directory names and ignores comments
$ RUN list "$CLITMP/vimrc"
foo
bar
$
# check reports package-manager and clone drift without mutation
$ VIM_PLUGIN_HOME="$CLITMP/vim" VIM_PLUG_RUNTIME="$CLITMP/missing/plug.vim" $VP --check "$CLITMP/vimrc"; echo "rc=$?"
  would: reconcile Vim plugins: vim-plug-package foo bar
rc=1
$ RUN --check "$CLITMP/vimrc"; echo "rc=$?"
  would: reconcile Vim plugins: foo bar
rc=1
$ test ! -e "$CLITMP/vim"; echo "rc=$?"
rc=0
$
# fake Vim records install/update actions and creates fallback plugin clones
$ printf '%s\n' '#!/usr/bin/env bash' 'printf "%s\n" "$*" > "$FAKE_LOG.vim"' 'case "$*" in *"PlugUpdate --sync"*) echo update > "$FAKE_LOG.action" ;; *) echo install > "$FAKE_LOG.action" ;; esac' 'mkdir -p "$VIM_PLUGIN_HOME/plugged/foo/.git" "$VIM_PLUGIN_HOME/plugged/bar/.git"' > "$CLITMP/bin/vim"; chmod +x "$CLITMP/bin/vim"; echo fake
fake
$ RUN apply "$CLITMP/vimrc" | sed "s|$CLITMP|TMP|g"
  installed fallback Vim plugins from TMP/vimrc
$ grep -o 'PlugInstall --sync' "$CLITMP/call.vim"
PlugInstall --sync
$ cat "$CLITMP/call.action"
install
$
# converged apply/check are no-ops; update routes through PlugUpdate --sync
$ RUN apply "$CLITMP/vimrc"
  already installed (packaged vim-plug + 2 fallback plugins)
$ RUN --check "$CLITMP/vimrc" >/dev/null; echo "rc=$?"
rc=0
$ RUN update "$CLITMP/vimrc" | sed "s|$CLITMP|TMP|g"
  updated fallback Vim plugins from TMP/vimrc
$ grep -o 'PlugUpdate --sync' "$CLITMP/call.vim"
PlugUpdate --sync
$ cat "$CLITMP/call.action"
update
$
# missing plugin and manager package paths fail clearly
$ rm -r "$CLITMP/vim/plugged/foo/.git"; RUN --check "$CLITMP/vimrc"; echo "rc=$?"
  would: reconcile Vim plugins: foo
rc=1
$ out=$(VIM_PLUGIN_HOME="$CLITMP/vim" VIM_PLUG_RUNTIME="$CLITMP/missing/plug.vim" $VP apply "$CLITMP/vimrc" 2>&1); rc=$?; printf '%s\n' "$out" | sed "s|$CLITMP|TMP|g"; echo "rc=$rc"
vim-plugins: vim-plug runtime missing (TMP/missing/plug.vim) — install the declared vim-plug package
rc=1
$
# malformed/unknown input fails loudly
$ printf 'set number\n' > "$CLITMP/empty-vimrc"; out=$($VP list "$CLITMP/empty-vimrc" 2>&1); rc=$?; printf '%s\n' "$out" | sed "s|$CLITMP|TMP|g"; echo "rc=$rc"
vim-plugins: no active Plug declarations in TMP/empty-vimrc
rc=1
$ $VP frobnicate "$CLITMP/vimrc" 2>&1; echo "rc=$?"
vim-plugins: unknown argument: frobnicate   (try: vim-plugins -h)
rc=1
$ rm -r "$CLITMP"; echo done
done
$
