# setup/vim-plugins — vim-plug pin, manifest, and apply/update/check behavior
# (Story 2.49). Host-independent: fake Vim mutates only $CLITMP; vim-plug is
# downloaded through a local file URL, with no network or live ~/.vim state.
#
# Run from the repo root: clitest tests/vim-plugins.clitest.txt

$ VP=.local/bin/setup/vim-plugins; CLITMP=$(mktemp -d); mkdir -p "$CLITMP/bin" "$CLITMP/source"; printf "%s\n" "Plug 'owner/foo'" "\"Plug 'ignored/nope'" "Plug 'owner/bar', {'rtp': 'vim/'}" > "$CLITMP/vimrc"; printf 'pinned vim-plug fixture\n' > "$CLITMP/source/plug.vim"; PSHA=$(sha256sum "$CLITMP/source/plug.vim" | cut -d' ' -f1); RUN() { PATH="$CLITMP/bin:$PATH" FAKE_LOG="$CLITMP/call" VIM_PLUGIN_HOME="$CLITMP/vim" VIM_PLUG_URL="file://$CLITMP/source/plug.vim" VIM_PLUG_SHA256="$PSHA" "$VP" "$@"; }; echo ready
ready
$
# list derives vim-plug's default directory names and ignores comments
$ RUN list "$CLITMP/vimrc"
foo
bar
$
# check reports the pinned manager and every missing clone; mutates nothing
$ RUN --check "$CLITMP/vimrc"; echo "rc=$?"
  would: reconcile Vim plugins: vim-plug foo bar
rc=1
$ test ! -e "$CLITMP/vim"; echo "rc=$?"
rc=0
$
# fake Vim records the generated minimal vimrc and creates declared clones for
# both install and update actions
$ printf '%s\n' '#!/usr/bin/env bash' 'printf "%s\n" "$*" > "$FAKE_LOG.vim"' 'while (($#)); do if [[ $1 == -Nu ]]; then cfg=$2; break; fi; shift; done' 'grep "^Plug " "$cfg" > "$FAKE_LOG.plugins"' 'case "$*" in *"PlugUpdate --sync"*) echo update > "$FAKE_LOG.action" ;; *) echo install > "$FAKE_LOG.action" ;; esac' 'mkdir -p "$VIM_PLUGGED_DIR/foo/.git" "$VIM_PLUGGED_DIR/bar/.git"' > "$CLITMP/bin/vim"; chmod +x "$CLITMP/bin/vim"; echo fake
fake
$ RUN apply "$CLITMP/vimrc" | sed "s|$CLITMP|TMP|g"
  installed pinned vim-plug (88e31471818e9a29a8a20a0ee61360cfd7bdc1cd)
  installed Vim plugins from TMP/vimrc
$ cmp "$CLITMP/source/plug.vim" "$CLITMP/vim/autoload/plug.vim"; echo "rc=$?"
rc=0
$ cat "$CLITMP/call.plugins"
Plug 'owner/foo'
Plug 'owner/bar', {'rtp': 'vim/'}
$ grep -o 'PlugInstall --sync' "$CLITMP/call.vim"
PlugInstall --sync
$ cat "$CLITMP/call.action"
install
$
# converged apply/check are no-ops; update routes through PlugUpdate --sync
$ RUN apply "$CLITMP/vimrc" | sed "s|$CLITMP|TMP|g"
  already installed (vim-plug + 2 plugins in TMP/vim/plugged)
$ RUN --check "$CLITMP/vimrc" >/dev/null; echo "rc=$?"
rc=0
$ RUN update "$CLITMP/vimrc" | sed "s|$CLITMP|TMP|g"
  updated Vim plugins from TMP/vimrc
$ grep -o 'PlugUpdate --sync' "$CLITMP/call.vim"
PlugUpdate --sync
$ cat "$CLITMP/call.action"
update
$
# manager checksum drift is red, then repaired atomically from the pin
$ printf tampered > "$CLITMP/vim/autoload/plug.vim"; RUN --check "$CLITMP/vimrc"; echo "rc=$?"
  would: reconcile Vim plugins: vim-plug
rc=1

plug.vim
