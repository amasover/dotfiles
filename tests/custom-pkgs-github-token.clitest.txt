# tools/custom-pkgs and the read-only GitHub API token. A curl stub records
# its argv and the header lines curl would read from stdin (`-H @-`); the
# token is a fake value that must reach only that header, never argv, the
# output or a `bash -x` trace. No network.
#
# Run from the repo root: clitest tests/custom-pkgs-github-token.clitest.txt

$ CP=$PWD/.local/bin/tools/custom-pkgs; T=$(mktemp -d); mkdir -p "$T/bin" "$T/home/.config/dotfiles" "$T/pkgbuilds/alpha"; printf 'pkgname=alpha\npkgver=1.0.0\npkgrel=1\nurl="https://github.com/acme/alpha"\n' > "$T/pkgbuilds/alpha/PKGBUILD"; echo ok
ok
$ printf '%s\n' '#!/bin/sh' 'echo "curl $*" >> "$T/argv"' 'cat > "$T/headers"' 'echo "{\"tag_name\":\"v2.0.0\",\"published_at\":\"2026-01-01T00:00:00Z\"}"' > "$T/bin/curl"; chmod +x "$T/bin/curl"; export T; CPX() { HOME="$T/home" CUSTOM_PKGS_DIR="$T/pkgbuilds" PATH="$T/bin:$PATH" "$@"; }; echo stub
stub
$
# no token file: the request stays anonymous (an empty header stream)
$ CPX "$CP" outdated | cut -f1-3; wc -c < "$T/headers" | tr -d ' '; sed "s|$T|T|g" "$T/argv"
alpha	1.0.0	2.0.0
0
curl -fsSL --max-time 15 -H @- https://api.github.com/repos/acme/alpha/releases/latest
$
# with a token: the header carries it, argv and the output never do
$ printf 'fake-token-123\n' > "$T/home/.config/dotfiles/github-token"; : > "$T/argv"; CPX "$CP" outdated > "$T/out" 2>&1; cat "$T/headers"; grep -c fake-token "$T/argv" "$T/out" | sed "s|$T|T|g"
Authorization: Bearer fake-token-123
T/argv:0
T/out:0
$
# not in a `bash -x` trace either
$ CPX bash -x "$CP" outdated 2> "$T/trace" > /dev/null; grep -c 'set +x' "$T/trace"; grep -c fake-token "$T/trace"
1
0
$
# a file without a trailing newline, or with a CRLF ending, still works
$ printf 'fake-token-123' > "$T/home/.config/dotfiles/github-token"; CPX "$CP" outdated > /dev/null; cat "$T/headers"
Authorization: Bearer fake-token-123
$ printf 'fake-token-123\r\n' > "$T/home/.config/dotfiles/github-token"; CPX "$CP" outdated > /dev/null; od -c "$T/headers" | grep -c '\\r'
0
$
# an empty file counts as no token
$ : > "$T/home/.config/dotfiles/github-token"; CPX "$CP" outdated > /dev/null; wc -c < "$T/headers" | tr -d ' '
0
$ rm -rf "$T"; echo done
done
$
