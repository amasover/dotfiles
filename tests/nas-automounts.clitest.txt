# NAS automount installation, bootstrap step 6f (Story 2.54, #249).
# Run from repo root: clitest tests/nas-automounts.clitest.txt
#
# Host-independent by construction: HOME, the /etc prefix, and every privileged
# or autofs-specific binary are redirected at a scratch tree, so this needs no
# NAS, no autofs package, and no root.

$ ATMP=$(mktemp -d); AHOME=$ATMP/home; AROOT=$ATMP/root; ABIN=$ATMP/bin; BARE=$ATMP/bare; SRC=$AHOME/.config/dotfiles/autofs; TRACE=$ATMP/systemctl-trace; CREDS=$ATMP/creds-written; export BOOTSTRAP_TEST_MODE=1 TRACE CREDS; mkdir -p "$AHOME" "$AROOT/etc" "$ABIN" "$BARE"; echo ready
ready
$ printf '%s\n' '#!/bin/sh' 'exec "$@"' > "$ABIN/sudo"; printf '%s\n' '#!/bin/sh' 'exit 0' > "$ABIN/automount"; printf '%s\n' '#!/bin/sh' 'printf "%s\n" "$*" >> "$TRACE"' 'case $1 in is-enabled) exit "${AUTOFS_ENABLED:-1}" ;; esac' > "$ABIN/systemctl"; chmod +x "$ABIN"/*; cp "$ABIN/sudo" "$ABIN/systemctl" "$BARE/"; ln -s "$(command -v bash)" "$BARE/bash"; echo stubs
stubs
$

  linked: auto.nas
