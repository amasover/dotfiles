# vm-autofit (Story 5.4, #151) — the resolution-hint reactor's testable seams.
# A stubbed xrandr on PATH records calls; no X server involved.

$ d=$(mktemp -d) && printf '#!/bin/sh\necho "xrandr $*" >> "$XLOG"\n' > "$d/xrandr" && chmod +x "$d/xrandr" && export XLOG="$d/calls.log" PATH="$d:$PATH"
$

apply with an explicit output targets it:

$ : > "$XLOG"; .local/bin/tools/vm-autofit apply Virtual-1; cat "$XLOG"
xrandr --output Virtual-1 --auto
$

apply without an output applies everything:

$ : > "$XLOG"; .local/bin/tools/vm-autofit apply; cat "$XLOG"
xrandr --auto
$

react applies once up front (a hint may already be pending), then once per
event line:

$ : > "$XLOG"; printf 'e1\ne2\n' | .local/bin/tools/vm-autofit react Virtual-1; wc -l < "$XLOG" | tr -d ' '
3
$

react on immediate EOF still catches the pending hint:

$ : > "$XLOG"; printf '' | .local/bin/tools/vm-autofit react; wc -l < "$XLOG" | tr -d ' '
1
$

unknown and missing subcommands are rejected loudly:

$ .local/bin/tools/vm-autofit bogus 2>&1; echo "rc=$?"
usage: vm-autofit apply|react|watch [output]
rc=2
$ .local/bin/tools/vm-autofit 2>&1; echo "rc=$?"
usage: vm-autofit apply|react|watch [output]
rc=2
$
