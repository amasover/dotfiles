# setup/openviking-memory — OpenViking clients for omp and Claude Code, wired
# from the per-machine server.env. Host-independent: a scratch HOME and stubs
# for curl, bao, git, npm and claude that log every argv to a trace; node is
# the real one. The key must never reach an argv, the trace or the output.
#
# Run from the repo root: clitest tests/openviking-memory.clitest.txt

$ t=$PWD/.local/bin/setup/openviking-memory; OVT=$(mktemp -d); S=$OVT/stub; export OVT TRACE=$OVT/trace HOME=$OVT/home PATH="$OVT/stub:$PATH"; KEY=ov-key-5b1f; mkdir -p "$S" "$HOME/.config/dotfiles/openviking"; cp .config/dotfiles/openviking/plugin.json "$HOME/.config/dotfiles/openviking/"; echo 0.5.0 > "$OVT/server-version"; echo "$KEY" > "$OVT/good-key"; echo "$KEY" > "$OVT/bao-key"; : > "$TRACE"; echo ready
ready
$ printf '%s\n' '#!/bin/sh' 'echo "curl $*" >> "$TRACE"' 'hdr=; case " $* " in *" -H @- "*) hdr=$(cat) ;; esac' 'for last; do :; done' 'case $last in' '  */health) [ -f "$OVT/down" ] && exit 7; printf "{\"status\":\"ok\",\"version\":\"%s\"}" "$(cat "$OVT/server-version")" ;;' '  */api/v1/stats/memories) [ "$hdr" = "X-API-Key: $(cat "$OVT/good-key")" ] || exit 22 ;;' '  */v1/auth/token/revoke-self) [ "$hdr" = "X-Vault-Token: tok-123" ] || exit 22; echo revoked >> "$TRACE" ;;' 'esac' > "$S/curl"
$ printf '%s\n' '#!/bin/sh' 'echo "bao $*" >> "$TRACE"' 'case $1 in' '  login) printf tok-123 ;;' '  kv) [ "$BAO_TOKEN" = tok-123 ] || exit 2; printf "%s\n" "$(cat "$OVT/bao-key")" ;;' 'esac' > "$S/bao"
$ printf '%s\n' '#!/bin/sh' 'echo "git $*" >> "$TRACE"' '[ "$1" = -C ] && exit 0' 'for d; do :; done' 'e=$d/examples' 'mkdir -p "$d/.git" "$e/pi-coding-agent-extension" "$e/memory-plugin-shared" "$e/claude-code-memory-plugin/.claude-plugin"' 'echo "export default {}" > "$e/pi-coding-agent-extension/index.ts"' 'echo "{\"version\":\"0.4.13\"}" > "$e/pi-coding-agent-extension/package.json"' ': > "$e/memory-plugin-shared/sync.mjs"' 'echo "{\"version\":\"0.7.6\"}" > "$e/claude-code-memory-plugin/.claude-plugin/plugin.json"' > "$S/git"
$ printf '%s\n' '#!/bin/sh' 'echo "npm $*" >> "$TRACE"' > "$S/npm"; printf '%s\n' '#!/bin/sh' 'echo "claude $*" >> "$TRACE"' 'case "$1 $2 $3" in' '  "plugin list ") [ -f "$OVT/mkt" ] && printf "  openviking-memory@openviking\n    Read from: %s/claude-code-memory-plugin\n" "$(cat "$OVT/mkt")" ;;' '  "plugin marketplace add") echo "$4" > "$OVT/mkt" ;;' '  "plugin marketplace remove") rm -f "$OVT/mkt" ;;' 'esac' 'exit 0' > "$S/claude"; chmod +x "$S"/*; echo stubs
stubs
$
